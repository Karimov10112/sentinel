Python

import pandas as pd, requests
df = pd.read_csv("samples/stream_20000.csv").head(500)
r = requests.post("http://127.0.0.1:5000/api/v1/flows?telegram=0",
                  data=df.to_csv(index=False),
                  headers={"Content-Type": "text/csv"})
print(r.json())
